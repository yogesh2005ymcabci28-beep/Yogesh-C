#include <stdio.h>

int main() {
    char str1[100];
    char str2[100];
    int length = 0;
    int i = 0, j = 0;
    int is_equal = 1; // 1 means true, 0 means false

    // 1. Read a string from the user (handles spaces using %[^\n])
    printf("Enter the first string: ");
    scanf("%[^\n]", str1);
    
    // Clear input buffer to read the next string safely
    while (getchar() != '\n'); 

    printf("Enter the second string: ");
    scanf("%[^\n]", str2);

    // 2. Calculate the length of the first string manually
    // Strings in C always end with the null terminator character: '\0'
    while (str1[length] != '\0') {
        length++;
    }
    printf("\n--- Results ---\n");
    printf("Length of the first string: %d\n", length);

    // 3. Reverse the first string in place
    printf("Reversed first string: ");
    for (i = length - 1; i >= 0; i--) {
        putchar(str1[i]);
    }
    printf("\n");

    // 4. Compare the two strings manually
    i = 0;
    while (str1[i] != '\0' || str2[i] != '\0') {
        if (str1[i] != str2[i]) {
            is_equal = 0; // Strings do not match
            break;
        }
        i++;
    }

    if (is_equal == 1) {
        printf("The two strings are identical.\n");
    } else {
        printf("The two strings are different.\n");
    }

    return 0;
}
