#include <stdio.h>

int main() {
    int numbers[5];
    int max;

    printf("Enter 5 integers:\n");
    for (int i = 0; i < 5; i++) {
        scanf("%d", &numbers[i]);
    }

    // Assume the first element is the largest initially
    max = numbers[0];

    // Compare with the rest of the elements
    for (int i = 1; i < 5; i++) {
        if (numbers[i] > max) {
            max = numbers[i];
        }
    }

    printf("The largest number is: %d\n", max);
    return 0;
}
