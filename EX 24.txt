#include <stdio.h>
#include <stdlib.h>

// Global pointers for tracking books and pages per shelf
int* total_number_of_books;
int** total_number_of_pages;

int main() {
    int total_number_of_shelves, total_number_of_queries;
    scanf("%d %d", &total_number_of_shelves, &total_number_of_queries);
    
    // Allocate tracking arrays
    total_number_of_books = (int*)calloc(total_number_of_shelves, sizeof(int));
    total_number_of_pages = (int**)calloc(total_number_of_shelves, sizeof(int*));
    
    while (total_number_of_queries--) {
        int type_of_query, x, y;
        scanf("%d", &type_of_query);
        
        if (type_of_query == 1) { // Add book
            scanf("%d %d", &x, &y);
            total_number_of_books[x]++;
            total_number_of_pages[x] = realloc(total_number_of_pages[x], total_number_of_books[x] * sizeof(int));
            total_number_of_pages[x][total_number_of_books[x] - 1] = y;
        } else if (type_of_query == 2) { // Print book
            scanf("%d %d", &x, &y);
            printf("%d\n", total_number_of_pages[x][y]);
        } else { // Print total books
            scanf("%d", &x);
            printf("%d\n", total_number_of_books[x]);
        }
    }
    
    // Free allocated memory
    for (int i = 0; i < total_number_of_shelves; i++) free(total_number_of_pages[i]);
    free(total_number_of_pages);
    free(total_number_of_books);
    
    return 0;
}
